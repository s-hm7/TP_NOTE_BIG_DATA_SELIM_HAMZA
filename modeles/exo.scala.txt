sc.setLogLevel("ERROR")

val entree = "hdfs/..."                       // dossier ou fichier (relatif à la racine du dépôt)
val lignes = sc.textFile(entree)
println("Lignes lues : " + lignes.count())

val res = lignes
  .filter(_.nonEmpty)                          // TODO 1 : filtre
  .map(l => (l.split(";")(0), 1))              // TODO 2 : (clé, valeur)
  .reduceByKey(_ + _)                          // TODO 3 : agrégat
  .sortBy(_._2, false)

res.take(20).foreach(println)
println("Nombre de clés : " + res.count())
